Table: ZoneDetails
Columns:
ZoneID int (PK)
ZoneName varchar (e.g. 'North', 'South', 'East', 'West')

Table: Regions
Columns:
RegionsId int (PK)
ZoneID int (FK -> ZoneDetails.ZoneID)
RegionsName varchar

Table: Locations
Columns:
LocationId int (PK)
ZoneID int (FK -> ZoneDetails.ZoneID)
RegionsId int (FK -> Regions.RegionsId)
LocationName varchar

Table: REAssets
Columns:
REAssetId int (PK)
AssetName varchar
DeveloperName varchar
BorrowerName varchar
ZoneId int (FK -> ZoneDetails.ZoneID)
RegionsId int (FK -> Regions.RegionsId)
LocationId int (FK -> Locations.LocationId)

Table: REUnitDetails
Columns:
REUnitDetailId int (PK)
REAssetId int (FK -> REAssets.REAssetId)
ProjectName varchar
DevelopmentType varchar
ProjectType varchar
Wing varchar
Floor varchar
UnitNumber varchar
Configuration varchar
AreaConsidered float
AreaConsideredMeasurement varchar (values: 'Sq Ft', 'Sq mtr', 'Sq yards')
UniqueKey varchar

Table: REPriceHeaders
Columns:
REPriceHeaderId int (PK)
REAssetID int (FK -> REAssets.REAssetId)
ValueType varchar (e.g. 'Sale Value')
HeaderValue varchar

Table: RESales
Columns:
RESalesID int (PK)
REAssetId int (FK -> REAssets.REAssetId)
REUnitDetailId varchar (FK -> REUnitDetails.UniqueKey)
CustomerName varchar
BookingDate date
RegistrationDate date

Table: REUnitSales
Columns:
REUnitSalesID int (PK)
RESalesID int (FK -> RESales.RESalesID)
PriceHeader varchar (FK -> REPriceHeaders.HeaderValue)
Amount float        <- IMPORTANT: float type
Demand money        <- IMPORTANT: money type
Collections money   <- IMPORTANT: money type

TYPE CAST RULE - CRITICAL:
PostgreSQL cannot mix float and money in arithmetic.
ALWAYS cast money columns to numeric before arithmetic:
  Collections::numeric
  Demand::numeric
Example correct receivables: SUM(ru.Amount) - SUM(ru.Collections::numeric)

JOIN RULES:
- REUnitSales.RESalesID = RESales.RESalesID
- REPriceHeaders.REAssetID = RESales.REAssetId AND REPriceHeaders.HeaderValue = REUnitSales.PriceHeader AND REPriceHeaders.ValueType = 'Sale Value'
- REAssets.ZoneId = ZoneDetails.ZoneID
- REAssets.RegionsId = Regions.RegionsId
- REAssets.LocationId = Locations.LocationId
- RESales.REUnitDetailId = REUnitDetails.UniqueKey AND RESales.REAssetId = REUnitDetails.REAssetId

CALCULATIONS:
Total Sales        = SUM(ru.Amount)
Balance Receivable = SUM(ru.Amount) - SUM(ru.Collections::numeric)
Area in Sq Ft      = SUM(CASE AreaConsideredMeasurement WHEN 'Sq Ft' THEN 1 * AreaConsidered WHEN 'Sq mtr' THEN 10.7639 * AreaConsidered WHEN 'Sq yards' THEN 9 * AreaConsidered END)

FILTER RULES:
- Sales/amount/receivable queries: always join REPriceHeaders with ValueType = 'Sale Value'
- Zone queries: join REAssets -> ZoneDetails on ZoneId = ZoneID
- Region queries: join REAssets -> Regions on RegionsId
- Sold units only: WHERE RESales.BookingDate IS NOT NULL
- Asset name search: AssetName ILIKE '%name%'
